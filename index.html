Python · core implementation
class LinkList:
    def __init__(self, capacity=4):
        self.capacity = int(capacity)
        self.free_pointer = 0
        self.start_pointer = -1
        self.list = [None] * capacity
        self.pointers = [i + 1 for i in range(capacity)]
        self.pointers[capacity - 1] = -1

    def insert(self, item):
        if self.free_pointer == -1:
            return False
        temp = self.start_pointer
        self.start_pointer = self.free_pointer
        self.free_pointer = self.pointers[self.free_pointer]
        self.pointers[self.start_pointer] = temp
        self.list[self.start_pointer] = item
        return True

    def find(self, item):
        temp = self.start_pointer
        while temp != -1 and self.list[temp] != item:
            temp = self.pointers[temp]
        return temp

    def delete(self, item):
        if self.start_pointer == -1:
            return False
        index = self.start_pointer
        while index != -1 and self.list[index] != item:
            old_pointer = index
            index = self.pointers[index]
        if index == -1:
            return False
        if index == self.start_pointer:
            self.start_pointer = self.pointers[self.start_pointer]
            self.pointers[index] = self.free_pointer
            self.free_pointer = index
            self.list[index] = None
        else:
            temp = self.pointers[index]
            self.pointers[index] = self.free_pointer
            self.free_pointer = index
            self.pointers[old_pointer] = temp
            self.list[index] = None
        return True

Python · stack operations
class Stack:
    def __init__(self, capacity):
        self.items = [None] * capacity
        self.top = -1

    def push(self, item):
        if self.top == len(self.items) - 1:
            return False
        self.top += 1
        self.items[self.top] = item
        return True

    def pop(self):
        if self.top == -1:
            return None
        item = self.items[self.top]
        self.items[self.top] = None
        self.top -= 1
        return item

    def peek(self):
        return None if self.top == -1 else self.items[self.top]

class Queue:
    def __init__(self, capacity):
        self.items = [None] * capacity
        self.front = 0
        self.rear = -1
        self.count = 0

    def enqueue(self, item):
        if self.count == len(self.items):
            return False
        self.rear = (self.rear + 1) % len(self.items)
        self.items[self.rear] = item
        self.count += 1
        return True

    def dequeue(self):
        if self.count == 0:
            return None
        item = self.items[self.front]
        self.items[self.front] = None
        self.front = (self.front + 1) % len(self.items)
        self.count -= 1
        if self.count == 0:
            self.front, self.rear = 0, -1
        return item

Python · bubble sort
def bubble_sort(values):
    n = len(values)
    for end in range(n - 1, 0, -1):
        swapped = False
        for i in range(end):
            if values[i] > values[i + 1]:
                values[i], values[i + 1] = values[i + 1], values[i]
                swapped = True
        if not swapped:
            break
    return values

def insertion_sort(values):
    for i in range(1, len(values)):
        key = values[i]
        j = i - 1
        while j >= 0 and values[j] > key:
            values[j + 1] = values[j]
            j -= 1
        values[j + 1] = key
    return values

Python · binary search
def binary_search(values, target):
    low, high = 0, len(values) - 1
    while low <= high:
        mid = (low + high) // 2
        if values[mid] == target:
            return mid
        if values[mid] < target:
            low = mid + 1
        else:
            high = mid - 1
    return -1

Python · recursive binary search
def binary_search(values, target, low, high):
    if low > high:
        return -1
    mid = (low + high) // 2
    if values[mid] == target:
        return mid
    if values[mid] < target:
        return binary_search(values, target, mid + 1, high)
    return binary_search(values, target, low, mid - 1)

class LinearQueue:
    def __init__(self, capacity):
        self.items = [None] * capacity
        self.front = 0
        self.rear = -1
        self.count = 0

    def enqueue(self, item):
        if self.rear == len(self.items) - 1:
            return False
        self.rear += 1
        self.items[self.rear] = item
        self.count += 1
        return True

    def dequeue(self):
        if self.count == 0:
            return None
        item = self.items[self.front]
        self.items[self.front] = None
        self.front += 1
        self.count -= 1
        return item

Python · linked-list insertion and deletion
class Node:
    def __init__(self, value):
        self.value = value
        self.prev = None
        self.next = None

class LinkedList:
    def insert_tail(self, value):
        node = Node(value)
        if self.head is None:
            self.head = self.tail = node
        else:
            node.prev = self.tail
            self.tail.next = node
            self.tail = node

    def delete(self, value):
        node = self.head
        while node is not None and node.value != value:
            node = node.next
        if node is None:
            return False
        if node.prev is not None:
            node.prev.next = node.next
        else:
            self.head = node.next
        if node.next is not None:
            node.next.prev = node.prev
        else:
            self.tail = node.prev
        return True

Python · linear probing
def hash_key(key, size):
    return sum(ord(char) for char in key) % size

def insert(table, key):
    start = hash_key(key, len(table))
    for offset in range(len(table)):
        index = (start + offset) % len(table)
        if table[index] is None:
            table[index] = key
            return index
        if table[index] == key:
            return index
    raise OverflowError("hash table is full")

Python · recursive functions
def factorial(n):
    if n <= 1:
        return 1
    return n * factorial(n - 1)

def fibonacci(n):
    if n <= 1:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)

def countdown(n):
    if n < 0:
        return
    print(n)
    countdown(n - 1)

def summation(n):
    if n <= 0:
        return 0
    return n + summation(n - 1)
